(* ─── Шаблонная функция ─── *)
объявление-шаблонной-функции = "template" "{" список-шаблонных-параметров "}"
		возвращаемый-тип имя-функции
		"<" список-специализаций-функции ">"
		"(" список-аргументов ")" ";"

определение-шаблонной-функции = "template" "{" список-шаблонных-параметров "}"
		возвращаемый-тип имя-функции
		"<" список-специализаций-функции ">"
		"(" список-аргументов ")"
		"{" тело-функции "}"

список-специализаций-функции = специализированный-параметр {"," специализированный-параметр}
специализированный-параметр = тип-специализации "." (имя-подтипа-специализации | "void")

вызов-шаблонной-функции = имя-функции
		"{" список-конкретных-типов "}"
		"<" список-указателей-на-специализаций ">"
		"(" список-аргументов ")"

список-указателей-на-специализаций = пусто | указатель-на-специализацию {"," указатель-на-специализацию}


(* ─── Шаблонная структура ─── *)
определение-шаблонной-структуры = "template" "{" список-шаблонных-параметров "}"
		"struct" имя-структуры "{" тело-структуры "}"
		"<" [список-специализаций] ">" ";"


добавление-специализаций = "template" "{" список-шаблонных-параметров "}"
		тип-специализации "{" список-шаблонных-параметров "}" "+" "<" список-специализаций ">"

список-специализаций = имя-типа ";" | список-групп-признаков
список-групп-признаков = группа-признаков ";" { группа-признаков ";"}
группа-признаков = признак {"," признак} ":" тип

объявление-переменной = ссылка-на-структуру "{" список-конкретных-типов "}" список-переменных ";"
список-переменных = переменная {"," переменная}
переменная = {"*"} имя-переменной {"[" размерность "]"} ["=" инициализатор]
ссылка-на-структуру = "struct" имя-структуры | typedef-имя


(* ─── Typedef ─── *)
объявление-typedef = полный-typedef-с-именем
		| полный-typedef-анонимный
		| краткий-typedef-структуры
		| краткий-typedef-типа

полный-typedef  = "template" "{" список-шаблонных-параметров "}"
		"typedef" "struct" имя-структуры
		"{" тело-структуры "}" typedef-имя ";"

полный-typedef-без-имени = "template" "{" список-шаблонных-параметров "}"
		"typedef" "struct"
		"{" тело-структуры "}" typedef-имя ";"

краткий-typedef-структуры = "typedef" "struct" имя-структуры typedef-имя ";"

краткий-typedef-типа = "typedef" имя-шаблонного-типа typedef-имя ";"

(* ─── Вспомогательные правила для параметров и аргументов ─── *)
список-шаблонных-параметров = шаблонный-параметр {"," шаблонный-параметр}
шаблонный-параметр = идентификатор

список-конкретных-типов = конкретный-тип {"," конкретный-тип}
конкретный-тип = идентификатор

список-аргументов = пусто | аргумент {"," аргумент}
аргумент = тип идентификатор

(* ─── Замена имен ─── *)
новое_имя = "template_"имя-шаблона"_"{"__"имя-типа}
имя-типа = идентификатор{"_"идентификатор}
